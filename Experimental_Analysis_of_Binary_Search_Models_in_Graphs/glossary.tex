% Key Definitions
\newglossaryentry{binarysearch}{
    name=Binary Search,
    description={A search strategy that repeatedly divides the search space by repeatedly querying the median vertex},
    nonumberlist
}

\newglossaryentry{decisiontree}{
    name=Decision Tree,
    description={A rooted tree whose nodes represent ordering of queries to be made depending on the previous responses},
    nonumberlist
}

\newglossaryentry{querycost}{
    name=Query Cost,
    description={The cost $c\br{v}$ of querying vertex $v$},
    nonumberlist
}

\newglossaryentry{searchcost}{
    name=Search Cost,
    description={The total cost of all queries performed to identify a target vertex},
    nonumberlist
}

\newglossaryentry{worstcase}{
    name=Worst Case,
    description={The maximum search cost over all possible target vertices},
    nonumberlist
}

\newglossaryentry{vertexranking}{
    name=Vertex Ranking,
    description={A labeling $l:V\to \brc{1,2,\dots,k}$ where for each pair $u,v$ with $l\br{u}=l\br{v}$, there exists $z$ on the path between $u$ and $v$ with $l\br{z}>l\br{v}$},
    nonumberlist
}

\newglossaryentry{upresponse}{
    name=Up Response,
    description={A response to query $v$ in rooted tree $T$ where the returned component contains the root $r\br{T}$},
    nonumberlist
}

\newglossaryentry{downresponse}{
    name=Down Response,
    description={A response to query $v$ in rooted tree $T$ where the returned component does not contain the root $r\br{T}$},
    nonumberlist
}

\newglossaryentry{heavyvertex}{
    name=Heavy Vertex,
    description={A vertex $v$ with query cost $c\br{v} > pk$ for some parameter $p$ and box size $k$},
    nonumberlist
}

\newglossaryentry{lightvertex}{
    name=Light Vertex,
    description={A vertex $v$ with query cost $c\br{v} \leq pk$ for some parameter $p$ and box size $k$},
    nonumberlist
}

\newglossaryentry{heavymodule}{
    name=Heavy Module,
    description={A maximal connected subset $H\subseteq V\br{T}$ where every vertex in $H$ is heavy},
    nonumberlist
}

\newglossaryentry{alignedcost}{
    name=Aligned Cost,
    description={A modified cost measure for aligned decision trees that ignores light queries with down responses},
    nonumberlist
}

\newglossaryentry{ptas}{
    name=PTAS,
    description={Polynomial-Time Approximation Scheme: $(1+\epsilon)$-approximation algorithm running in polynomial time for fixed $\epsilon$},
    nonumberlist
}

\newglossaryentry{eptas}{
    name=EPTAS,
    description={Efficient Polynomial-Time Approximation Scheme: PTAS with running time $f\br{\epsilon}\cdot \text{poly}\br{n}$ where $f$ depends only on $\epsilon$},
    nonumberlist
}

\newglossaryentry{qptas}{
    name=QPTAS,
    description={Quasi-Polynomial-Time Approximation Scheme: $(1+\epsilon)$-approximation algorithm running in $n^{O\br{\text{polylog}\br{n}}}$ time},
    nonumberlist
}

\newglossaryentry{fptas}{
    name=FPTAS,
    description={Fully Polynomial-Time Approximation Scheme: PTAS with running time $O\br{\text{poly}\br{n, 1/\epsilon}}$ time},
    nonumberlist
}

\newglossaryentry{approximationratio}{
    name=Approximation Ratio,
    description={The ratio $\frac{\text{ALG}}{\text{OPT}}$ between algorithm's solution and optimal solution},
    nonumberlist
}

\newglossaryentry{path}{
    name=Path,
    description={A sequence of vertices where each adjacent pair is connected by an edge},
    nonumberlist
}

\newglossaryentry{cycle}{
    name=Cycle,
    description={A closed path where the first and last vertices are the same},
    nonumberlist
}

\newglossaryentry{forest}{
    name=Forest,
    description={An acyclic graph; a disjoint union of trees},
    nonumberlist
}

\newglossaryentry{connectedcomponent}{
    name=Connected Component,
    description={A maximal connected subgraph where every pair of vertices has a path between them},
    nonumberlist
}

\newglossaryentry{rootedtree}{
    name=Rooted Tree,
    description={A tree with a designated root vertex $r$ that establishes parent-child relationships},
    nonumberlist
}

\newglossaryentry{inducedsubgraph}{
    name=Induced Subgraph,
    description={For $S\subseteq V\br{G}$, the subgraph $G\angl{S}$ with vertex set $S$ and all edges from $E\br{G}$ with both endpoints in $S$},
    nonumberlist
}

\newglossaryentry{subtree}{
    name=Subtree,
    description={A connected subgraph of a tree},
    nonumberlist
}
