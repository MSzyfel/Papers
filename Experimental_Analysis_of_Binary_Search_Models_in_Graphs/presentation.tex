\section{Problem Definition}

\begin{frame}{The Tree Search Problem}
\begin{block}{Problem Statement}
Given a tree $T = (V, E)$ with a hidden target vertex $x$:
\begin{itemize}
    \item Query a vertex $v$ with cost $c(v)$
    \item If $v = x$: target found
    \item Otherwise: receive the subtree containing $x$
    \item Goal: Minimize worst-case total query cost
\end{itemize}
\end{block}

\pause

\begin{block}{Problem Variants}
\begin{itemize}
    \item \textbf{Uniform costs:} $T||V||\COST_{max}$ -- all queries have unit cost
    \item \textbf{Non-uniform costs:} $T||V, c||\COST_{max}$ -- queries have arbitrary costs
\end{itemize}
\end{block}
\end{frame}

\section{Related Work}

\subsection{Three-Field Notation}

\begin{frame}{Three-Field Notation}
\begin{block}{Notation: $\alpha||\beta||\gamma$}
Classification scheme for search problems:
\begin{itemize}
    \item $\alpha$ -- \textbf{Search domain:} structure being searched
    \begin{itemize}
        \item $T$ (trees), $G$ (graphs), $P$ (posets)
    \end{itemize}
    \item $\beta$ -- \textbf{Target set and costs:}
    \begin{itemize}
        \item $V$ -- any vertex can be target
        \item $c$ -- non-uniform query costs (omitted if uniform)
    \end{itemize}
    \item $\gamma$ -- \textbf{Optimization criterion:}
    \begin{itemize}
        \item $\COST_{max}$ -- minimize worst-case cost
        \item $\COST_{avg}$ -- minimize average cost
    \end{itemize}
\end{itemize}
\end{block}

\vspace{0.3cm}
\textbf{Examples:}
\begin{itemize}
    \item $T||V||\COST_{max}$ -- uniform costs, worst-case optimization
    \item $T||V, c||\COST_{max}$ -- non-uniform costs, worst-case optimization
\end{itemize}
\end{frame}

\begin{frame}{Cost Function Generation}
\begin{block}{Cost Distributions Tested}
\begin{enumerate}
    \item \textbf{Uniform costs ($c \sim \mathcal{U}(0,1)$):}
    \begin{itemize}
        \item Continuous uniform distribution on $[0, 1]$
        \item All costs equally likely in the interval
    \end{itemize}
    
    \pause
    
    \item \textbf{Binomial costs ($c \sim \mathcal{B}(n=10, p=0.5)$):}
    \begin{itemize}
        \item Discrete distribution with $n=10$ trials, $p=0.5$
        \item Costs concentrate around mean value
    \end{itemize}
    
    \pause
    
    \item \textbf{Exponential costs ($c \sim \mathcal{E}(\lambda=1)$):}
    \begin{itemize}
        \item Continuous distribution with rate $\lambda = 1$
        \item Many small costs, few large costs
    \end{itemize}
\end{enumerate}
\end{block}

\end{frame}

\begin{frame}{Measured Performance Metrics}
\begin{block}{For All Algorithms}
\begin{itemize}
    \item \textbf{Average Runtime:} Mean execution time over all instances of size $n$
    \item \textbf{Maximum Runtime:} Worst-case execution time for size $n$
\end{itemize}
\end{block}

\pause

\begin{block}{For Uniform Cost Algorithms}
\begin{itemize}
    \item \textbf{Average Depth:} Mean depth of decision trees produced
    \item \textbf{Maximum Depth:} Maximum depth observed
\end{itemize}
\end{block}

\pause

\begin{block}{For Non-Uniform Cost Algorithms}
\begin{itemize}
    \item \textbf{Average Approximation Ratio:} Mean of $\frac{\text{ALG}}{\OPT}$
    \item \textbf{Maximum Approximation Ratio:} Worst ratio observed
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{RankingBasedDT Performance}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{Key Findings:}
\begin{itemize}
    \item Runtime: Near-linear for random and bounded degree trees
    \item Runtime: $O(n \log n)$ for bounded diameter
    \item Depth: Logarithmic for random/bounded degree
    \item Depth: Constant (≤4) for bounded diameter
\end{itemize}

\pause

\column{0.5\textwidth}
\textbf{Observations:}
\begin{itemize}
    \item Bounded diameter: trivial upper bound $\lceil \frac{\text{diam}(T)+1}{2} \rceil$
    \item Bounded degree: trivial lower bound $\log_{\Delta}n$
    \item Conjecture: For random trees: $\mathbb{E}[\text{rank}(T)] = \Theta(\log n)$ or w.h.p. $\text{rank}(T) = \Theta(\log n)$
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Exact Algorithm Runtime Analysis}
\begin{block}{Exponential function regression: $T(n) = b^n$}
Fitted bases $b$ for different tree types:
\begin{itemize}
    \item Random trees: $b \approx 1.34 - 1.39$
    \item Bounded degree: $b \approx 1.35 - 1.38$ 
    \item Bounded diameter: $b \approx 1.51 - 1.61$
\end{itemize}
\end{block}

\pause

\textbf{Practical implications:}
\begin{itemize}
    \item Much better than worst-case $O(2^n)$
    \item Solved instances up to $n \approx 33$ vertices
    \item Matches theoretical bound $O(1.42^n)$ w.h.p. for random trees
\end{itemize}
\end{frame}

\begin{frame}{Approximation Algorithms: Quality}
\begin{block}{Approximation Ratios}
All algorithms achieved:
\begin{itemize}
    \item Average ratios: $< 2.0$ across all configurations
    \item Maximum ratios: $< 3.0$ in worst cases
    \item No significant difference between algorithms
\end{itemize}
\end{block}

\pause

\textbf{Surprising observation:}
\begin{itemize}
    \item $O(\log n / \log \log n)$-approximation often performed \textit{best}
    \item More greedy approach works better on random instances
    \item $O(\log \log n)$-approximation slightly worse due to multiple cost levels
\end{itemize}
\end{frame}

\begin{frame}{Approximation Algorithms: Runtime}
\begin{block}{Practical Usability}
\begin{itemize}
    \item $O(\log n / \log \log n)$-approximation: practical for $n \leq 100$
    \begin{itemize}
        \item Stable runtime, no spikes
        \item Recommended for practical use
    \end{itemize}
    \item $O(\sqrt{\log n})$ and $O(\log \log n)$: limited to small instances ($n \leq 50$)
    \begin{itemize}
        \item High variance in runtime
        \item QPTAS bottleneck when $\text{rank}(T)$ is large
    \end{itemize}
\end{itemize}
\end{block}

\pause

\textbf{QPTAS behavior:}
\begin{itemize}
    \item Runtime: $n^{O(\log \text{rank}(T) / \epsilon^2)}$
    \item Path-like auxiliary trees cause slowdowns
    \item For bounded diameter: becomes PTAS with $n^{O(D/\epsilon^2)}$ time
\end{itemize}
\end{frame}
